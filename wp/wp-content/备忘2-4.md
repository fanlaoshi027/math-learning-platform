# 四、Access Service核心类

文件：


作用：

所有权限判断入口。

------

代码：

```php
<?php

namespace MathCourse\Access;

defined('ABSPATH') || exit;


class Access_Service
{


/**
 * 用户是否拥有课程权限
 */
public function has_course_access(
$user_id,
$course_id
)
{


if(
!$user_id ||
!$course_id
){

return false;

}


//管理员直接通过

if(
user_can(
$user_id,
'manage_options'
)
){

return true;

}



return $this->check_grant(
$user_id,
$course_id
);


}



/**
 * 查询授权
 */
private function check_grant(
$user_id,
$course_id
)
{


global $wpdb;


$table =
$wpdb->prefix .
'mathcourse_access';



$row =
$wpdb->get_row(

$wpdb->prepare(

"
SELECT *

FROM $table

WHERE user_id=%d

AND course_id=%d

AND status=%s

",

$user_id,

$course_id,

'active'

)

);



if(
!$row
){

return false;

}



//检查有效期

if(
$row->expire_time
&&
strtotime($row->expire_time)
<
time()

){

return false;

}



return true;


}



}
```

# 三、Token类设计

文件：

```
includes/Video/class-token.php
```

作用：

生成、验证播放授权。

------

代码：

```php
<?php

namespace MathCourse\Video;

defined('ABSPATH') || exit;


class Token
{


/**
 * 生成播放Token
 */
public function create(
$video_id,
$user_id,
$expire = 1800
)
{


$expire_time =
time() + $expire;


$data =
$video_id .
'|' .
$user_id .
'|' .
$expire_time;



$signature =
hash_hmac(
'sha256',
$data,
AUTH_KEY
);



return array(

'expire'=>$expire_time,

'token'=>$signature

);


}



/**
 * 验证Token
 */
public function verify(
$video_id,
$user_id,
$expire,
$token
)
{


if(
time() > $expire
){

return false;

}



$data =
$video_id .
'|' .
$user_id .
'|' .
$expire;



$signature =
hash_hmac(
'sha256',
$data,
AUTH_KEY
);



return hash_equals(
$signature,
$token
);


}


}
```

```php
__NAMESPACE__ . '\\';

$class_name =
str_replace(
$prefix,
'',
$class

str_replace(

$class_name
);



$file =
strtolower(
preg_replace(
'/([a-z])([A-Z])/',
'$1-$2',
$file
)
);



$path =
MATHCOURSE_PATH .
'includes/' .
$file .
'.php';
```
