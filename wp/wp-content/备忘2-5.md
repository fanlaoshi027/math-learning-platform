# 10.2 Token安全类

文件：

```text
includes/Video/class-token.php
```

作用：

生成临时播放授权。

------

设计：

有效期：

默认30分钟。

------

代码：

```php
<?php

namespace MathCourse\Video;


defined('ABSPATH') || exit;


class Token
{


public function create(
$video_id,
$user_id
)
{


$expire =
time()+1800;



$data =
$video_id .
'|' .
$user_id .
'|' .
$expire;



$sign =
hash_hmac(

'sha256',

$data,

AUTH_KEY

);



return array(

'expire'=>$expire,

'token'=>$sign

);


}





public function verify(
$video_id,
$user_id,
$expire,
$token
)
{


if(
time()>$expire
){

return false;

}



$data =
$video_id .
'|' .
$user_id .
'|' .
$expire;



$sign =
hash_hmac(

'sha256',

$data,

AUTH_KEY

);



return hash_equals(

$sign,

$token

);


}


}
```

  # 11.2 Lesson权限控制

  文件：

  ```text
  includes/Tutor/class-lesson-access.php
  ```

  作用：

  判断：

  当前Lesson是否允许学习。

  ------

  逻辑：

  ```text
  Lesson
  
  ↓
  
  是否试听？
  
  是
  
  ↓
  
  允许
  
  
  否
  
  ↓
  
  检查课程授权
  
  
  有
  
  ↓
  
  允许
  
  
  无
  
  ↓
  
  拒绝
  ```

  ------

  代码：

  ```php
  <?php
  
  namespace MathCourse\Tutor;
  
  
  defined('ABSPATH') || exit;
  
  
  class Lesson_Access
  {
  
  
  public function check()
  {
  
  
  $lesson_id =
  get_the_ID();
  
  
  
  /*
  试听判断
  */
  
  if(
  $this->is_preview(
  $lesson_id
  )
  ){
  
  return true;
  
  }
  
  
  
  /*
  获取课程ID
  
  */
  
  $course_id =
  $this->get_course_id(
  $lesson_id
  );
  
  
  
  if(
  !$course_id
  ){
  
  return false;
  
  }
  
  
  
  $user_id =
  get_current_user_id();
  
  
  
  $service =
  new \MathCourse\Access\Access_Service();
  
  
  
  return $service->has_course_access(
  
  $user_id,
  
  $course_id
  
  );
  
  
  }
  
  
  
  
  
  private function is_preview(
  $lesson_id
  )
  {
  
  
  return (bool)
  get_post_meta(
  
  $lesson_id,
  
  '_lesson_preview',
  
  true
  
  );
  
  
  }
  
  
  
  
  
  private function get_course_id(
  $lesson_id
  )
  {
  
  
  /*
  
  Tutor LMS关联课程
  
  */
  
  return tutor_utils()
  ->get_course_id_by(
  'lesson',
  $lesson_id
  );
  
  
  }
  
  
  }
  ```

```php
<?php

namespace MathCourse;


defined('ABSPATH') || exit;


class Autoloader
{


public static function register()
{

spl_autoload_register(
array(
__CLASS__,
'load'
)
);

}



public static function load($class)
{


$prefix='MathCourse\\';



if(
strpos($class,$prefix)!==0
){

return;

}



$class =
str_replace(
$prefix,
'',
$class
);



$parts =
explode(
'\\',
$class
);



$class_name =
array_pop($parts);



$folder =
implode(
'/',
$parts
);



$file =
'class-' .
self::kebab($class_name)
.
'.php';



$path =
MATHCOURSE_PATH .
'includes/' .
$folder .
'/' .
$file;



if(
file_exists($path)
){

require_once $path;

}


}





private static function kebab($name)
{


return strtolower(

preg_replace(
'/([a-z])([A-Z])/',
'$1-$2',
$name
)

);


}



}
```

```php
<?php

namespace MathCourse\Video;


defined('ABSPATH') || exit;



class Token
{


private $expire_time = 1800;



public function create(
$video_id,
$user_id
)
{


$expire =
time()+$this->expire_time;



$data =
$video_id .
'|' .
$user_id .
'|' .
$expire;



$token =
hash_hmac(

'sha256',

$data,

AUTH_KEY

);



return array(

'expire'=>$expire,

'token'=>$token

);


}




public function verify(
$video_id,
$user_id,
$expire,
$token
)
{


if(
time()>intval($expire)
){

return false;

}



$data =
$video_id .
'|' .
$user_id .
'|' .
$expire;



$check =
hash_hmac(

'sha256',

$data,

AUTH_KEY

);



return hash_equals(

$check,

$token

);


}



}
```

```php
class Complete_Service
{

$lesson_id
)
{


global $wpdb;


$table =
$wpdb->prefix .
'mathcourse_learning';



return $wpdb->replace(

$table,

array(

'user_id'=>$user_id,

'course_id'=>$course_id,

'lesson_id'=>$lesson_id,

'completed_time'=>current_time('mysql')

),

array(

'%d',

'%d',

'%d',

'%s'

)

);

public function is_complete(
$user_id,
$lesson_id
)
{


global $wpdb;


$table =
$wpdb->prefix .
'mathcourse_learning';



$id =
$wpdb->get_var(

$wpdb->prepare(

"
SELECT id

FROM {$table}

WHERE user_id=%d

AND lesson_id=%d

LIMIT 1

$user_id,

$lesson_id

)

);
```
