<?php

namespace MathCourse\Models;

defined('ABSPATH') || exit;


class Lesson
{


    public static function save($data)
    {


        global $wpdb;


        $table =
        $wpdb->prefix .
        'mathcourse_lessons';



        $wpdb->insert(

            $table,

            array(

                'tutor_lesson_id'=>
                intval(
                    $data['tutor_lesson_id']
                ),


                'course_id'=>
                intval(
                    $data['course_id']
                ),


                'title'=>
                sanitize_text_field(
                    $data['title']
                ),


                'video_type'=>
                'hls',


                'video_url'=>
                esc_url_raw(
                    $data['video_url']
                ),


                'video_encrypt'=>
                sanitize_text_field(
                    $data['encrypt']
                ),


                'pdf_url'=>
                esc_url_raw(
                    $data['pdf_url']
                ),


                'page_start'=>
                intval(
                    $data['page_start']
                ),


                'page_end'=>
                intval(
                    $data['page_end']
                ),


                'knowledge_points'=>
                sanitize_textarea_field(
                    $data['knowledge']
                ),


                'difficulty'=>
                sanitize_text_field(
                    $data['difficulty']
                )

            )

        );



        return $wpdb->insert_id;


    }


}
