# 三、新增视频授权类

创建：

```
includes/Video/class-token.php
```

代码：

```php
<?php

namespace MathCourse\Video;

defined('ABSPATH') || exit;


class Token
{


    private static $secret =
    'mathcourse-secret-key';



    public static function create(
        $path,
        $user_id,
        $expire=1800
    )
    {


        $time =
        time()+$expire;



        $sign =
        hash_hmac(

            'sha256',

            $path.
            '|'.
            $user_id.
            '|'.
            $time,

            self::$secret

        );



        return array(

            'token'=>$sign,

            'expire'=>$time

        );


    }





    public static function verify(
        $path,
        $user_id,
        $expire,
        $token
    )
    {


        if(
            time()>$expire
        ){

            return false;

        }



        $check =
        hash_hmac(

            'sha256',

            $path.
            '|'.
            $user_id.
            '|'.
            $expire,

            self::$secret

        );



        return hash_equals(
            $check,
            $token
        );


    }



}
```

```php
<?php

namespace MathCourse\Ajax;

defined('ABSPATH') || exit;


class Progress
{


public function register()
{


add_action(
'wp_ajax_mathcourse_progress',
array(
$this,
'save'
)
);


}



public function save()
{


$user_id =
get_current_user_id();



if(!$user_id){

wp_die();

}



$lesson_id =
intval(
$_POST['lesson_id']
);



$current =
intval(
$_POST['current']
);



$duration =
intval(
$_POST['duration']
);



$percent = 0;


if($duration>0){

$percent =
round(
$current/$duration*100
);

}



global $wpdb;


$table =
$wpdb->prefix.
'mathcourse_progress';



$wpdb->replace(

$table,

array(

'user_id'=>$user_id,

'lesson_id'=>$lesson_id,

'video_time'=>$current,

'video_duration'=>$duration,

'watch_percent'=>$percent

)

);



wp_send_json_success(

array(

'percent'=>$percent

)

);



}



}
```

```php
<?php


namespace MathCourse;


defined('ABSPATH') || exit;



class Autoloader
{


public static function register()
{


spl_autoload_register(

array(
__CLASS__,
'load'
)

);


}



public static function load(
$class
)
{


$prefix =
'MathCourse\\';



if(
strpos(
$class,
$prefix
)!==0
){

return;

}



$file =
str_replace(
$prefix,
'',
$class
);



$file =
str_replace(
'\\',
'/',
$file
);



$file =
MATHCOURSE_PATH .
'includes/' .
'class-' .
strtolower(
$file
)
.
'.php';



if(
file_exists($file)
){

require_once $file;

}


}



}



Autoloader::register();
```
